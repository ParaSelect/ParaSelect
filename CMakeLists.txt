cmake_minimum_required(VERSION 3.26)

set(PARASELECT_CUDA_ROOT "" CACHE PATH "CUDA toolkit root (empty: infer from CUDACXX or nvcc on PATH)")

if(NOT CMAKE_CUDA_COMPILER)
    if(DEFINED ENV{CUDACXX})
        set(CMAKE_CUDA_COMPILER "$ENV{CUDACXX}" CACHE FILEPATH "CUDA compiler" FORCE)
    elseif(PARASELECT_CUDA_ROOT AND EXISTS "${PARASELECT_CUDA_ROOT}/bin/nvcc")
        set(CMAKE_CUDA_COMPILER "${PARASELECT_CUDA_ROOT}/bin/nvcc" CACHE FILEPATH "CUDA compiler" FORCE)
    else()
        find_program(_paraselect_nvcc NAMES nvcc HINTS ENV CUDA_PATH ENV CUDA_HOME)
        if(_paraselect_nvcc)
            set(CMAKE_CUDA_COMPILER "${_paraselect_nvcc}" CACHE FILEPATH "CUDA compiler" FORCE)
        else()
            message(FATAL_ERROR "CUDA nvcc was not found. Set CUDACXX, PARASELECT_CUDA_ROOT, or add nvcc to PATH.")
        endif()
    endif()
endif()

if(NOT PARASELECT_CUDA_ROOT)
    get_filename_component(_paraselect_cuda_bin "${CMAKE_CUDA_COMPILER}" DIRECTORY)
    get_filename_component(_paraselect_cuda_root "${_paraselect_cuda_bin}" DIRECTORY)
    set(PARASELECT_CUDA_ROOT "${_paraselect_cuda_root}" CACHE PATH "CUDA toolkit root" FORCE)
endif()
if(NOT CUDAToolkit_ROOT)
    set(CUDAToolkit_ROOT "${PARASELECT_CUDA_ROOT}" CACHE PATH "CUDA Toolkit root")
endif()

project(paraselect_tests LANGUAGES CXX CUDA)
enable_testing()
option(PARASELECT_SKIP_TESTS "Skip all test targets" OFF)

# Use Release mode by default.
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
